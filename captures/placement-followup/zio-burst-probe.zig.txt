// Append to zio src/runtime.zig (zio 40a53a7) and run:
//   zig build test -Dscheduling=pinned -Doptimize=ReleaseFast
// It fails on purpose (error.ProbeDone) so the runner prints its lines.
// Executor 1 is used because executor 0 runs only while the main thread
// blocks inside zio; a test that parks the main thread in an OS sleep
// freezes executor 0 (an earlier version measured exactly that).
test "probe: spawn burst under a held mutex on a busy pinned executor" {
    if (zio_options.scheduling != .pinned or builtin.single_threaded) return error.SkipZigTest;
    const Mutex = @import("sync/Mutex.zig");
    const H = struct {
        const busy_n = 300;
        const burst = 250;
        var work: u32 = 50000;
        fn busy(stop: *std.atomic.Value(bool), bticks: *std.atomic.Value(u64)) void {
            var x: u64 = 0;
            while (!stop.load(.acquire)) {
                sleep(.fromMilliseconds(1)) catch return;
                _ = bticks.fetchAdd(1, .monotonic);
                var i: u32 = 0;
                while (i < work) : (i += 1) x +%= i *% 2654435761;
                std.mem.doNotOptimizeAway(x);
            }
        }
        fn child(mu: *Mutex, stop: *std.atomic.Value(bool), ticks: *std.atomic.Value(u64), started: *std.atomic.Value(u32)) void {
            mu.lockUncancelable();
            mu.unlock();
            _ = started.fetchAdd(1, .monotonic);
            while (!stop.load(.acquire)) {
                sleep(.fromMilliseconds(1)) catch return;
                _ = ticks.fetchAdd(1, .monotonic);
            }
        }
        fn actor(mu: *Mutex, group: *Group, stop: *std.atomic.Value(bool), ticks: *std.atomic.Value(u64), started: *std.atomic.Value(u32)) !void {
            mu.lockUncancelable();
            defer mu.unlock();
            for (0..burst) |_| try group.spawnInto(.local, child, .{ mu, stop, ticks, started });
        }
    };
    var round: u32 = 0;
    while (round < 3) : (round += 1) {
        const runtime = try Runtime.init(std.testing.allocator, .{ .executors = .exact(2) });
        defer runtime.deinit();
        var stop = std.atomic.Value(bool).init(false);
        var ticks = std.atomic.Value(u64).init(0);
        var started = std.atomic.Value(u32).init(0);
        var bticks = std.atomic.Value(u64).init(0);
        var mu: Mutex = .init;
        var busy: Group = .init;
        defer busy.cancel();
        var kids: Group = .init;
        defer kids.cancel();
        for (0..H.busy_n) |_| try busy.spawnInto(.{ .executor = 1 }, H.busy, .{ &stop, &bticks });
        os.time.sleep(.fromMilliseconds(50));
        var a = try runtime.spawnInto(.{ .executor = 1 }, H.actor, .{ &mu, &kids, &stop, &ticks, &started });
        try a.join();
        var t: u32 = 0;
        while (t < 10) : (t += 1) {
            os.time.sleep(.fromMilliseconds(500));
            std.debug.print("probe round {d} t={d}ms: children started {d}/{d}, child ticks {d}, busy ticks {d}\n", .{ round, (t + 1) * 500, started.load(.acquire), H.burst, ticks.load(.acquire), bticks.load(.acquire) });
        }
        stop.store(true, .release);
        try busy.wait();
        try kids.wait();
    }
    return error.ProbeDone;
}
